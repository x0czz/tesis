load_dataset <- function(dataset_name) {
  candidates <- c(
    file.path(base_dir, paste0(dataset_name, ".sav")),
    file.path(base_dir, paste0(dataset_name, ".RData"))
  )

  for (file_path in candidates) {
    if (!file.exists(file_path)) next

    if (grepl("\\.sav$", file_path, ignore.case = TRUE)) {
      return(read_sav(file_path))
    }

    env <- new.env(parent = emptyenv())
    load(file_path, envir = env)
    objs <- ls(env)
    if (length(objs) == 0L) stop("El archivo RData está vacío: ", file_path)
    return(get(objs[[1]], envir = env))
  }

  stop("No se encontró el archivo para ", dataset_name, ". Busqué .sav y .RData en: ", base_dir)
}

# ---- Casos completos por gráfico ----
casos_completos <- function(df, vars) {
  df %>%
    group_by(id_estudiante) %>%
    filter(
      n_distinct(ola) == 2,
      all(if_all(all_of(vars), ~ !is.na(.x)))
    ) %>%
    ungroup()
}

# ---- 1. Distribución de respuestas por ola (barras 100%) ----
dist_cat <- function(df, vars, items, etq, colores, titulo) {
  tab <- casos_completos(df, vars) %>%
    select(ola, all_of(vars)) %>%
    pivot_longer(-ola, names_to = "item", values_to = "resp") %>%
    filter(!is.na(resp)) %>%
    count(item, ola, resp) %>%
    group_by(item, ola) %>%
    mutate(N = sum(n), pct = n / N) %>%
    ungroup() %>%
    mutate(item = factor(item, levels = vars, labels = items),
           resp = factor(resp, levels = seq_along(etq), labels = etq),
           eje = paste0(ola, " (n=", N, ")"),
           oscuro = as.integer(resp) <= ceiling(length(etq) / 2))
  ggplot(tab, aes(x = pct, y = eje, fill = resp)) +
    geom_col(width = 0.62, colour = "white", linewidth = 0.5,
             position = position_stack(reverse = TRUE)) +
    geom_text(aes(label = ifelse(pct >= 0.05, percent(pct, accuracy = 1), ""),
                  colour = oscuro),
              position = position_stack(vjust = 0.5, reverse = TRUE), size = 3) +
    scale_colour_manual(values = c("TRUE" = tinta, "FALSE" = "#ffffff"), guide = "none") +
    scale_fill_manual(values = setNames(colores, etq),
                      guide = guide_legend(nrow = if (length(etq) > 4) 2 else 1)) +
    scale_x_continuous(labels = percent, expand = c(0, 0)) +
    scale_y_discrete(limits = rev) +
    facet_wrap(~ item, ncol = 1, scales = "free_y",
               labeller = labeller(item = label_wrap_gen(60))) +
    labs(title = titulo, subtitle = "% de respuestas válidas", x = NULL, y = NULL)
}

# ---- 2. "Serie de tiempo": media e IC 95% de Ola 1 a Ola 2 ----
media_ola <- function(df, vars, items, titulo, por = NULL, ylim = c(1, 4)) {
  vars_completas <- c(vars, por)
  datos <- casos_completos(df, vars_completas)
  datos$grupo <- if (is.null(por)) "Total" else as.character(datos[[por]])
  datos %>%
    filter(!is.na(grupo)) %>%
    select(ola, grupo, all_of(vars)) %>%
    pivot_longer(all_of(vars), names_to = "item", values_to = "x") %>%
    filter(!is.na(x)) %>%
    group_by(item, grupo, ola) %>%
    summarise(n = n(), media = mean(x), ee = sd(x) / sqrt(n), .groups = "drop") %>%
    mutate(item = factor(item, levels = vars, labels = items)) %>%
    ggplot(aes(x = ola, y = media, colour = ola)) +
    geom_path(aes(group = interaction(item, grupo)), colour = gris, linewidth = 0.6) +
    geom_errorbar(aes(ymin = media - 1.96 * ee, ymax = media + 1.96 * ee),
                  width = 0.1, linewidth = 0.6) +
    geom_point(size = 3) +
    geom_text(aes(label = sprintf("%.2f", media),
                  hjust = ifelse(ola == "Ola 1", 1.6, -0.6)),
              size = 3, colour = tinta2) +
    scale_colour_manual(values = col_ola, guide = "none") +
    scale_x_discrete(expand = expansion(add = 0.9)) +
    scale_y_continuous(limits = ylim) +
    facet_grid(item ~ grupo, labeller = labeller(item = label_wrap_gen(25))) +
    labs(title = titulo, subtitle = "Media e IC 95% por ola", x = NULL, y = "Media")
}

# ---- 3. Trayectoria individual Ola 1 -> Ola 2 (matriz de transición) ----
transicion <- function(df, v, k, titulo) {
  tab <- df %>%
    mutate(x = .data[[v]]) %>%
    select(id_estudiante, ola, x) %>%
    pivot_wider(names_from = ola, values_from = x) %>%
    filter(!is.na(`Ola 1`), !is.na(`Ola 2`))
  n_pares <- nrow(tab)
  tab %>%
    count(`Ola 1`, `Ola 2`) %>%
    group_by(`Ola 1`) %>%
    mutate(pct = n / sum(n)) %>%
    ungroup() %>%
    ggplot(aes(x = factor(`Ola 2`, levels = 1:k),
               y = factor(`Ola 1`, levels = k:1), fill = pct)) +
    geom_tile(colour = "white", linewidth = 1) +
    geom_text(aes(label = percent(pct, accuracy = 1), colour = pct > 0.5), size = 3.2) +
    scale_colour_manual(values = c("TRUE" = "#ffffff", "FALSE" = tinta), guide = "none") +
    scale_fill_gradient(low = "#cde2fb", high = "#104281", limits = c(0, 1),
                        labels = percent, name = "% de la fila") +
    coord_equal() +
    labs(title = titulo,
         subtitle = paste0("Estudiantes con dato en ambas olas (n = ", n_pares,
                           "). Cada fila suma 100%. 1 = muy en desacuerdo, 4 = muy de acuerdo"),
         x = "Ola 2", y = "Ola 1")
}

# ---- Asociación ordinal: Spearman con IC (aprox. de Fisher) y tau-b ----
asoc <- function(x, y, min_n = 10) {
  ok <- complete.cases(x, y); x <- x[ok]; y <- y[ok]; n <- length(x)
  if (n < min_n || sd(x) == 0 || sd(y) == 0)
    return(tibble(n = n, rho = NA_real_, ic_inf = NA_real_, ic_sup = NA_real_,
                  p = NA_real_, tau_b = NA_real_))
  ct <- suppressWarnings(cor.test(x, y, method = "spearman", exact = FALSE))
  r <- unname(ct$estimate); se <- 1.06 / sqrt(n - 3)
  tibble(n = n, rho = r, ic_inf = tanh(atanh(r) - 1.96 * se),
         ic_sup = tanh(atanh(r) + 1.96 * se), p = ct$p.value,
         tau_b = suppressWarnings(cor(x, y, method = "kendall")))
}

# ---- Spearman parcial (x, y controlando z) vía residuos de rangos ----
parcial <- function(x, y, z) {
  ok <- complete.cases(x, y, z)
  rx <- rank(x[ok]); ry <- rank(y[ok]); rz <- rank(z[ok])
  asoc(resid(lm(rx ~ rz)), resid(lm(ry ~ rz)))
}

# ---- Gráfico de efectos (rho con IC) ----
forest <- function(tab, titulo, facet = NULL) {
  p <- tab %>%
    mutate(par = paste0(hip, ": ", x, " ~ ", y)) %>%
    ggplot(aes(x = rho, y = par, colour = ola, xmin = ic_inf, xmax = ic_sup)) +
    geom_vline(xintercept = 0, colour = gris, linewidth = 0.4) +
    geom_linerange(orientation = "y", position = position_dodge(width = 0.6), linewidth = 0.6) +
    geom_point(position = position_dodge(width = 0.6), size = 2.8) +
    scale_colour_manual(values = col_ola) +
    scale_y_discrete(limits = rev) +
    labs(title = titulo, subtitle = "Spearman con IC 95% (aproximación de Fisher)",
         x = "rho", y = NULL)
  if (!is.null(facet)) p <- p + facet_wrap(facet, nrow = 1)
  p
}

w_ic <- function(k, n, lado, z = 1.96) {
  p <- k / n
  centro <- (p + z^2 / (2 * n)) / (1 + z^2 / n)
  mitad  <- z * sqrt(p * (1 - p) / n + z^2 / (4 * n^2)) / (1 + z^2 / n)
  if (lado == "lo") centro - mitad else centro + mitad
}

    group_by(ola, dv, nivel) |>

    mutate(p = k / n, lo = w_ic(k, n, "lo"), hi = w_ic(k, n, "hi"))

  ggplot(r, aes(nivel, p, colour = ola, group = ola)) +
    geom_linerange(aes(ymin = lo, ymax = hi), position = position_dodge(.35), linewidth = .5) +
    geom_line(position = position_dodge(.35), linewidth = .5, alpha = .7) +
    geom_point(aes(size = n), position = position_dodge(.35)) +
    scale_colour_manual(values = col_ola, name = NULL) +
    scale_size_area(max_size = 5, guide = "none") +
    scale_y_continuous(labels = scales::percent, limits = c(0, 1)) +
    facet_wrap(~dv) +
    labs(title = titulo, x = xlab %||% x, y = "% de acuerdo", caption = caption_n)
}

# ---------- 4. Predictor -> distribución completa de la VD (100% apilado) ----------
cruce_apilado <- function(d, x, titulo, xlab = NULL, dvs = c("p1_1", "p1_2")) {
  r <- largo_dv(d, x, dvs) |>
    count(ola, dv, nivel, y) |>
    group_by(ola, dv, nivel) |>
    mutate(pct = n / sum(n), ntot = sum(n)) |>
    ungroup() |>
    mutate(y = factor(y, levels = 1:4, labels = etq_lik))

  ggplot(r, aes(nivel, pct, fill = y)) +
    geom_col(width = .8, colour = "white", linewidth = .2) +
    scale_fill_manual(values = setNames(rampa4, etq_lik), name = NULL) +
    scale_y_continuous(labels = scales::percent, expand = c(0, 0)) +
    facet_grid(dv ~ ola) +
    labs(title = titulo, x = xlab %||% x, y = NULL,
         caption = "Distribución completa de la respuesta (escala ordinal de 4 puntos). Sin pruebas de hipótesis.") +
    guides(fill = guide_legend(nrow = 1))
}

# ---------- 5. Predictor continuo individual (ej. J-score) -> % de acuerdo por tramos ----------
acuerdo_tramos <- function(d, x, titulo, cortes = c(-Inf, -.5, -.1, .1, .5, Inf),
                           etq = c("Muy sub-recomp.", "Sub-recomp.", "Justo", "Sobre-recomp.", "Muy sobre-recomp."),
                           xlab = NULL) {
  d2 <- d |> mutate(.t = cut(.data[[x]], cortes, labels = etq))
  acuerdo_segun(d2, ".t", titulo, xlab = xlab %||% x)
}

# ---------- 6. Contextual (colegio): dispersión por ola ----------
disp_ctx <- function(d, x_ctx, titulo, xlab = NULL, unidad = "colegio1", dvs = c("p1_1", "p1_2")) {
  r <- d |>
    select(ola, u = all_of(unidad), xc = all_of(x_ctx), all_of(dvs)) |>
    pivot_longer(all_of(dvs), names_to = "dv", values_to = "y") |>
    filter(y %in% 1:4, !is.na(xc)) |>
    group_by(ola, dv, u) |>
    summarise(x = mean(xc, na.rm = TRUE), n = n(), p = mean(y >= 3), .groups = "drop") |>
    mutate(dv = factor(dv_lab[dv], levels = dv_lab))

  ggplot(r, aes(x, p, colour = ola)) +
    geom_smooth(method = "lm", se = FALSE, linewidth = .5, linetype = "dashed", alpha = .6) +
    geom_point(aes(size = n), alpha = .85) +
    geom_text(aes(label = u), size = 2.8, vjust = -1, show.legend = FALSE) +
    scale_colour_manual(values = col_ola, name = NULL) +
    scale_size_area(max_size = 6, guide = "none") +
    scale_y_continuous(labels = scales::percent, limits = c(0, 1)) +
    facet_wrap(~dv) +
    labs(title = titulo, x = xlab %||% x_ctx, y = "% de acuerdo (promedio del colegio)",
         caption = "Un punto = un colegio (n = 12). Tamaño = n de estudiantes. Línea punteada: tendencia lineal descriptiva.")
}

# ---------- 7. Panel: cambio del predictor vs cambio de la VD (base ancha w) ----------
cruce_cambios <- function(w, x, titulo, xlab = NULL, dvs = c("p1_1", "p1_2")) {
  cl <- function(v) case_when(v < 0 ~ "Baja", v == 0 ~ "Igual", v > 0 ~ "Sube")
  r <- map_dfr(dvs, function(y) {
    w |>
      transmute(
        cx = cl(.data[[paste0(x, "_w02")]] - .data[[paste0(x, "_w01")]]),
        cy = cl(.data[[paste0(y, "_w02")]] - .data[[paste0(y, "_w01")]]),
        dv = dv_lab[[y]]
      ) |>
      filter(!is.na(cx), !is.na(cy))
  }) |>
    count(dv, cx, cy) |>
    group_by(dv, cx) |>
    mutate(pct = n / sum(n), ntot = sum(n)) |>
    ungroup() |>
    mutate(cx = factor(cx, levels = c("Baja", "Igual", "Sube")),
           cy = factor(cy, levels = c("Baja", "Igual", "Sube")),
           dv = factor(dv, levels = dv_lab))

  ggplot(r, aes(cx, pct, fill = cy)) +
    geom_col(width = .75, colour = "white") +
    geom_text(aes(x = cx, y = 1.04, label = paste0("n=", ntot)), size = 2.8, colour = carbon,
              data = distinct(r, dv, cx, ntot), inherit.aes = FALSE) +
    scale_fill_manual(values = c(Baja = carbon, Igual = "#E6E6E6", Sube = rojo),
                      name = "Cambio en la VD (Ola 2 − Ola 1)") +
    scale_y_continuous(labels = scales::percent) +
    facet_wrap(~dv) +
    labs(title = titulo, x = paste0("Cambio en ", xlab %||% x, " (Ola 2 − Ola 1)"), y = NULL,
         caption = "Solo estudiantes presentes en ambas olas.")
}
